   443→STEP 5 — Core Library: Scoring Engine (lib/scoring-engine.ts)
   444→
   445→This is the critical logic file. Implement it completely:
   446→
   447→typescript// lib/scoring-engine.ts
   448→import { PrismaClient } from '@prisma/client';
   449→import type { RecordBallInput, RecordBallResponse } from '@/types';
   450→import { calculateCRR, calculateRRR, decimalOvers } from './scoring-utils';
   451→
   452→export async function recordBall(
   453→  prisma: PrismaClient,
   454→  inningsId: string,
   455→  input: RecordBallInput
   456→): Promise<RecordBallResponse> {
   457→  const innings = await prisma.innings.findUniqueOrThrow({
   458→    where: { id: inningsId },
   459→    include: {
   460→      match: true,
   461→      batting: true,
   462→      bowling: true,
   463→      balls: { orderBy: { deliveryNumber: 'desc' }, take: 1 },
   464→    },
   465→  });
   466→
   467→  const {
   468→    batsmanId, bowlerId, runs, isWicket, wicketType,
   469→    dismissedPlayerId, fielderPlayerId, extraType, extraRuns,
   470→  } = input;
   471→
   472→  const isWide = extraType === 'WIDE';
   473→  const isNoBall = extraType === 'NO_BALL';
   474→  const isBye = extraType === 'BYE';
   475→  const isLegBye = extraType === 'LEG_BYE';
   476→  const isLegalDelivery = !isWide && !isNoBall;
   477→  const totalRuns = runs + extraRuns;
   478→
   479→  // runs credited to bowler (byes/legbyes are NOT bowler's fault)
   480→  const runsAgainstBowler = (isBye || isLegBye) ? 0 : totalRuns;
   481→
   482→  const currentLegalBalls = innings.currentBalls;
   483→  const newBallInOver = isLegalDelivery ? currentLegalBalls + 1 : 0;
   484→  const isOverComplete = isLegalDelivery && newBallInOver === 6;
   485→
   486→  const lastDelivery = innings.balls[0];
   487→  const nextDeliveryNumber = (lastDelivery?.deliveryNumber ?? 0) + 1;
   488→
   489→  // ── 1. Create Ball record ─────────────────────────────────
   490→  const ball = await prisma.ball.create({
   491→    data: {
   492→      inningsId,
   493→      overNumber: innings.completedOvers,
   494→      ballInOver: isLegalDelivery ? newBallInOver : 0,
   495→      deliveryNumber: nextDeliveryNumber,
   496→      batsmanId,
   497→      bowlerId,
   498→      runs,
   499→      isWicket,
   500→      wicketType: isWicket ? wicketType : null,
   501→      dismissedPlayerId: isWicket ? dismissedPlayerId : null,
   502→      fielderPlayerId: (isWicket && fielderPlayerId) ? fielderPlayerId : null,
   503→      extraType: extraType ?? null,
   504→      extraRuns,
   505→      isLegalDelivery,
   506→      strikerIdBefore: innings.strikerId!,
   507→      nonStrikerIdBefore: innings.nonStrikerId!,
   508→    },
   509→  });
   510→
   511→  // ── 2. Update BatsmanInnings (striker) ───────────────────
   512→  const batsmanDismissed = isWicket && (
   513→    wicketType !== 'RUN_OUT'
   514→      ? true
   515→      : dismissedPlayerId === batsmanId
   516→  );
   517→
   518→  await prisma.batsmanInnings.upsert({
   519→    where: { inningsId_playerId: { inningsId, playerId: batsmanId } },
   520→    create: {
   521→      inningsId, playerId: batsmanId, battingOrder: 99,
   522→      runs: runs,
   523→      balls: isWide ? 0 : 1,
   524→      fours: runs === 4 && !isBye && !isLegBye ? 1 : 0,
   525→      sixes: runs === 6 && !isBye && !isLegBye ? 1 : 0,
   526→      isOut: batsmanDismissed,
   527→      dismissalType: batsmanDismissed ? wicketType : null,
   528→      dismissedByBowlerId: batsmanDismissed && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? bowlerId : null,
   529→      fielderPlayerId: batsmanDismissed && fielderPlayerId ? fielderPlayerId : null,
   530→    },
   531→    update: {
   532→      runs: { increment: runs },
   533→      balls: { increment: isWide ? 0 : 1 },
   534→      fours: { increment: runs === 4 && !isBye && !isLegBye ? 1 : 0 },
   535→      sixes: { increment: runs === 6 && !isBye && !isLegBye ? 1 : 0 },
   536→      ...(batsmanDismissed ? {
   537→        isOut: true,
   538→        dismissalType: wicketType,
   539→        dismissedByBowlerId: !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? bowlerId : null,
   540→        fielderPlayerId: fielderPlayerId ?? null,
   541→      } : {}),
   542→    },
   543→  });
   544→
   545→  // Handle run-out of non-striker
   546→  if (isWicket && wicketType === 'RUN_OUT' && dismissedPlayerId === innings.nonStrikerId) {
   547→    await prisma.batsmanInnings.upsert({
   548→      where: { inningsId_playerId: { inningsId, playerId: innings.nonStrikerId! } },
   549→      create: {
   550→        inningsId, playerId: innings.nonStrikerId!, battingOrder: 99,
   551→        isOut: true, dismissalType: 'RUN_OUT',
   552→      },
   553→      update: { isOut: true, dismissalType: 'RUN_OUT' },
   554→    });
   555→  }
   556→
   557→  // ── 3. Update BowlerInnings ───────────────────────────────
   558→  const currentBowler = innings.bowling.find(b => b.playerId === bowlerId);
   559→  const bowlerCurrentBalls = currentBowler?.balls ?? 0;
   560→  const bowlerNewBalls = isLegalDelivery ? (bowlerCurrentBalls + 1) % 6 : bowlerCurrentBalls;
   561→  const bowlerOverComplete = isLegalDelivery && (bowlerCurrentBalls + 1) === 6;
   562→
   563→  await prisma.bowlerInnings.upsert({
   564→    where: { inningsId_playerId: { inningsId, playerId: bowlerId } },
   565→    create: {
   566→      inningsId, playerId: bowlerId,
   567→      completedOvers: bowlerOverComplete ? 1 : 0,
   568→      balls: bowlerNewBalls,
   569→      runs: runsAgainstBowler,
   570→      wickets: isWicket && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? 1 : 0,
   571→      wides: isWide ? 1 : 0,
   572→      noBalls: isNoBall ? 1 : 0,
   573→    },
   574→    update: {
   575→      balls: bowlerNewBalls,
   576→      runs: { increment: runsAgainstBowler },
   577→      ...(bowlerOverComplete ? { completedOvers: { increment: 1 } } : {}),
   578→      ...(isWicket && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? { wickets: { increment: 1 } } : {}),
   579→      ...(isWide ? { wides: { increment: 1 } } : {}),
   580→      ...(isNoBall ? { noBalls: { increment: 1 } } : {}),
   581→    },
   582→  });
   583→
   584→  // ── 4. Update Innings state ───────────────────────────────
   585→  const newCurrentBalls = isOverComplete ? 0 : (isLegalDelivery ? currentLegalBalls + 1 : currentLegalBalls);
   586→  const newCompletedOvers = innings.completedOvers + (isOverComplete ? 1 : 0);
   587→  const newWickets = innings.wickets + (isWicket ? 1 : 0);
   588→  const newRuns = innings.runs + totalRuns;
   589→
   590→  // ── 5. Calculate new striker ─────────────────────────────
   591→  let newStrikerId = innings.strikerId!;
   592→  let newNonStrikerId = innings.nonStrikerId!;
   593→
   594→  if (isLegalDelivery) {
   595→    if (runs % 2 === 1) {
   596→      [newStrikerId, newNonStrikerId] = [newNonStrikerId, newStrikerId];
   597→    }
   598→    if (isOverComplete) {
   599→      // At end of over: if even runs, must also rotate (non-striker faces next over)
   600→      if (runs % 2 === 0) {
   601→        [newStrikerId, newNonStrikerId] = [newNonStrikerId, newStrikerId];
   602→      }
   603→    }
   604→  }
   605→
   606→  // If striker dismissed, non-striker takes strike position while new batsman awaits
   607→  if (batsmanDismissed) {
   608→    newStrikerId = newNonStrikerId;
   609→    newNonStrikerId = ''; // will be filled by new batsman selection
   610→  }
   611→
   612→  // ── 6. Persist innings state ──────────────────────────────
   613→  await prisma.innings.update({
   614→    where: { id: inningsId },
   615→    data: {
   616→      runs: newRuns,
   617→      wickets: newWickets,
   618→      completedOvers: newCompletedOvers,
   619→      currentBalls: newCurrentBalls,
   620→      wideBalls: { increment: isWide ? 1 : 0 },
   621→      noBalls: { increment: isNoBall ? 1 : 0 },
   622→      byes: { increment: isBye ? extraRuns : 0 },
   623→      legByes: { increment: isLegBye ? extraRuns : 0 },
   624→      strikerId: newStrikerId || null,
   625→      nonStrikerId: newNonStrikerId || null,
   626→    },
   627→  });
   628→
   629→  // ── 7. Check completion ────────────────────────────────────
   630→  const match = innings.match;
   631→  const isInningsComplete =
   632→    newWickets >= match.maxWickets ||
   633→    (newCompletedOvers >= match.totalOvers && newCurrentBalls === 0) ||
   634→    (innings.inningsNumber === 2 && innings.target != null && newRuns >= innings.target);
   635→
   636→  const ballsRemaining = innings.inningsNumber === 2 && innings.target != null
   637→    ? (match.totalOvers * 6) - (newCompletedOvers * 6 + newCurrentBalls)
   638→    : null;
   639→
   640→  return {
   641→    ball: ball as any,
   642→    inningsState: {
   643→      runs: newRuns,
   644→      wickets: newWickets,
   645→      completedOvers: newCompletedOvers,
   646→      currentBalls: newCurrentBalls,
   647→      currentRunRate: calculateCRR(newRuns, newCompletedOvers, newCurrentBalls),
   648→      requiredRunRate: innings.target
   649→        ? calculateRRR(innings.target - newRuns, match.totalOvers, newCompletedOvers, newCurrentBalls)
   650→        : null,
   651→      runsNeeded: innings.target ? innings.target - newRuns : null,
   652→      ballsRemaining,
   653→      isCompleted: isInningsComplete,
   654→      isOverComplete,
   655→    },
   656→    strikerUpdate: { strikerId: newStrikerId, nonStrikerId: newNonStrikerId },
   657→    needsNewBatsman: isWicket && !isInningsComplete,
   658→    needsNewBowler: isOverComplete && !isInningsComplete,
   659→    needsInningsBreak: isInningsComplete && innings.inningsNumber === 1,
   660→    isMatchComplete: isInningsComplete && innings.inningsNumber === 2,
   661→  };
   662→}

   664→export async function undoLastBall(
   665→  prisma: PrismaClient,
   666→  inningsId: string
   667→): Promise<{ success: boolean }> {
   668→  // 1. Get last ball
   669→  const lastBall = await prisma.ball.findFirst({
   670→    where: { inningsId },
   671→    orderBy: { deliveryNumber: 'desc' },
   672→  });
   673→  if (!lastBall) return { success: false };
   674→
   675→  const innings = await prisma.innings.findUniqueOrThrow({ where: { id: inningsId }, include: { match: true } });
   676→
   677→  const isWide = lastBall.extraType === 'WIDE';
   678→  const isNoBall = lastBall.extraType === 'NO_BALL';
   679→  const isBye = lastBall.extraType === 'BYE';
   680→  const isLegBye = lastBall.extraType === 'LEG_BYE';
   681→  const totalRuns = lastBall.runs + lastBall.extraRuns;
   682→  const runsAgainstBowler = (isBye || isLegBye) ? 0 : totalRuns;
   683→
   684→  // 2. Reverse BatsmanInnings
   685→  await prisma.batsmanInnings.update({
   686→    where: { inningsId_playerId: { inningsId, playerId: lastBall.batsmanId } },
   687→    data: {
   688→      runs: { decrement: lastBall.runs },
   689→      balls: { decrement: isWide ? 0 : 1 },
   690→      fours: { decrement: lastBall.runs === 4 && !isBye && !isLegBye ? 1 : 0 },
   691→      sixes: { decrement: lastBall.runs === 6 && !isBye && !isLegBye ? 1 : 0 },
   692→      ...(lastBall.isWicket ? { isOut: false, dismissalType: null, dismissedByBowlerId: null, fielderPlayerId: null } : {}),
   693→    },
   694→  });
   695→
   696→  // Reverse non-striker run-out
   697→  if (lastBall.isWicket && lastBall.wicketType === 'RUN_OUT' && lastBall.dismissedPlayerId === lastBall.nonStrikerIdBefore) {
   698→    await prisma.batsmanInnings.update({
   699→      where: { inningsId_playerId: { inningsId, playerId: lastBall.nonStrikerIdBefore } },
   700→      data: { isOut: false, dismissalType: null },
   701→    });
   702→  }
   703→
   704→  // 3. Reverse BowlerInnings
   705→  const bowlerInnings = await prisma.bowlerInnings.findUnique({
   706→    where: { inningsId_playerId: { inningsId, playerId: lastBall.bowlerId } },
   707→  });
   708→  if (bowlerInnings) {
   709→    const wasOverBall = lastBall.isLegalDelivery && lastBall.ballInOver === 6;
   710→    await prisma.bowlerInnings.update({
   711→      where: { inningsId_playerId: { inningsId, playerId: lastBall.bowlerId } },
   712→      data: {
   713→        runs: { decrement: runsAgainstBowler },
   714→        balls: wasOverBall ? 0 : { decrement: lastBall.isLegalDelivery ? 1 : 0 },
   715→        ...(wasOverBall ? { completedOvers: { decrement: 1 }, balls: 5 } : {}),
   716→        ...(lastBall.isWicket && !['RUN_OUT','RETIRED_HURT'].includes(lastBall.wicketType ?? '') ? { wickets: { decrement: 1 } } : {}),
   717→        ...(isWide ? { wides: { decrement: 1 } } : {}),
   718→        ...(isNoBall ? { noBalls: { decrement: 1 } } : {}),
   719→      },
   720→    });
   721→  }
   722→
   723→  // 4. Reverse Innings
   724→  const wasOverComplete = lastBall.isLegalDelivery && lastBall.ballInOver === 6;
   725→  const restoredBalls = wasOverComplete ? 5 : (lastBall.isLegalDelivery ? Math.max(0, innings.currentBalls - 1) : innings.currentBalls);
   726→  const restoredCompletedOvers = innings.completedOvers - (wasOverComplete ? 1 : 0);
   727→
   728→  await prisma.innings.update({
   729→    where: { id: inningsId },
   730→    data: {
   731→      runs: { decrement: totalRuns },
   732→      wickets: { decrement: lastBall.isWicket ? 1 : 0 },
   733→      completedOvers: restoredCompletedOvers,
   734→      currentBalls: restoredBalls,
   735→      wideBalls: { decrement: isWide ? 1 : 0 },
   736→      noBalls: { decrement: isNoBall ? 1 : 0 },
   737→      byes: { decrement: isBye ? lastBall.extraRuns : 0 },
   738→      legByes: { decrement: isLegBye ? lastBall.extraRuns : 0 },
   739→      strikerId: lastBall.strikerIdBefore,
   740→      nonStrikerId: lastBall.nonStrikerIdBefore,
   741→    },
   742→  });
   743→
   744→  // 5. Delete Ball record
   745→  await prisma.ball.delete({ where: { id: lastBall.id } });
   746→
   747→  return { success: true };
   748→}

   755→export function decimalOvers(completedOvers: number, balls: number): number {
   756→  return completedOvers + balls / 6;

   759→export function calculateCRR(runs: number, completedOvers: number, balls: number): number {
   760→  const ov = decimalOvers(completedOvers, balls);
   761→  if (ov === 0) return 0;
   762→  return Math.round((runs / ov) * 100) / 100;
   763→}
   764→
   765→export function calculateRRR(
   766→  runsNeeded: number,
   767→  totalOvers: number,
   768→  completedOvers: number,
   769→  balls: number
   770→): number {
   771→  const ballsTotal = totalOvers * 6;
   772→  const ballsUsed = completedOvers * 6 + balls;
   773→  const ballsLeft = ballsTotal - ballsUsed;
   774→  if (ballsLeft <= 0) return runsNeeded > 0 ? 99.99 : 0;
   775→  const oversLeft = ballsLeft / 6;
   776→  return Math.round((runsNeeded / oversLeft) * 100) / 100;
   777→}

   803→export function calculateNRR(
   804→  runsScored: number,
   805→  oversFaced: number,
   806→  runsConceded: number,
   807→  oversBowled: number
   808→): number {
   809→  const rpo_scored = oversFaced > 0 ? runsScored / oversFaced : 0;
   810→  const rpo_conceded = oversBowled > 0 ? runsConceded / oversBowled : 0;
   811→  return Math.round((rpo_scored - rpo_conceded) * 1000) / 1000;
   812→}

   814→export function generateResultString(
   815→  battingFirstTeamName: string,
   816→  chasingTeamName: string,
   817→  inn1Runs: number,
   818→  inn2Runs: number,
   819→  inn2Wickets: number,
   820→  inn2Completed: boolean,
   821→  inn2Overs: number,
   822→  inn2CurrentBalls: number,
   823→  totalOvers: number,
   824→  maxWickets: number
   825→): string {
   826→  if (inn2Runs >= inn1Runs + 1) {
   827→    const wicketsRemaining = maxWickets - inn2Wickets;
   828→    const ballsUsed = inn2Overs * 6 + inn2CurrentBalls;
   829→    const ballsTotal = totalOvers * 6;
   830→    const ballsRemaining = ballsTotal - ballsUsed;
   831→    if (ballsRemaining > 0) {
   832→      return `${chasingTeamName} won by ${wicketsRemaining} wicket${wicketsRemaining !== 1 ? 's' : ''} (${ballsRemaining} balls remaining)`;
   833→    }
   834→    return `${chasingTeamName} won by ${wicketsRemaining} wicket${wicketsRemaining !== 1 ? 's' : ''}`;
   835→  }
   836→  if (inn2Runs < inn1Runs + 1) {
   837→    return `${battingFirstTeamName} won by ${inn1Runs - inn2Runs} run${inn1Runs - inn2Runs !== 1 ? 's' : ''}`;
   838→  }
   839→  return `Match tied — both teams scored ${inn1Runs}`;
   840→}

   842→export function getManOfMatch(batting: { player: { name: string }; runs: number; balls: number }[], bowling: { player: { name: string }; wickets: number; runs: number }[]): string {
   843→  // Wickets priority: if any bowler took 3+, they're MoM
   844→  const topBowler = [...bowling].sort((a, b) => b.wickets - a.wickets || a.runs - b.runs)[0];
   845→  if (topBowler && topBowler.wickets >= 3) return topBowler.player.name;
   846→  const topBatsman = [...batting].sort((a, b) => b.runs - a.runs)[0];
   847→  return topBatsman?.player?.name ?? 'N/A';
   848→}

   850→export function generateRoundRobinSchedule(teamIds: string[]): [string, string][] {
   851→  const n = teamIds.length;
   852→  const teams = n % 2 === 0 ? [...teamIds] : [...teamIds, 'BYE'];
   853→  const rounds: [string, string][] = [];
   854→  const t = teams.length;
   855→  const pivot = teams[0];
   856→  const rotatable = teams.slice(1);
   857→  for (let round = 0; round < t - 1; round++) {
   858→    const current = [pivot, ...rotatable];
   859→    for (let i = 0; i < t / 2; i++) {
   860→      if (current[i] !== 'BYE' && current[t - 1 - i] !== 'BYE') {
   861→        rounds.push([current[i], current[t - 1 - i]]);
   862→      }
   863→    }
   864→    rotatable.unshift(rotatable.pop()!);
   865→  }
   866→  return rounds;
   867→}
