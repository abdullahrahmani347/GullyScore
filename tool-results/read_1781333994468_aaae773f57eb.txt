   259→   259→   758→   758→   758→
   260→   260→   759→   759→   759→export function calculateCRR(runs: number, completedOvers: number, balls: number): number {
   261→   261→   760→   760→   760→  const ov = decimalOvers(completedOvers, balls);
   262→   262→   761→   761→   761→  if (ov === 0) return 0;
   263→   263→   762→   762→   762→  return Math.round((runs / ov) * 100) / 100;
   264→   264→   763→   763→   763→}
   265→   265→   764→   764→   764→
   266→   266→   765→   765→   765→export function calculateRRR(
   267→   267→   766→   766→   766→  runsNeeded: number,
   268→   268→   767→   767→   767→  totalOvers: number,
   269→   269→   768→   768→   768→  completedOvers: number,
   270→   270→   769→   769→   769→  balls: number
   271→   271→   770→   770→   770→): number {
   272→   272→   771→   771→   771→  const ballsTotal = totalOvers * 6;
   273→   273→   772→   772→   772→  const ballsUsed = completedOvers * 6 + balls;
   274→   274→   773→   773→   773→  const ballsLeft = ballsTotal - ballsUsed;
   275→   275→   774→   774→   774→  if (ballsLeft <= 0) return runsNeeded > 0 ? 99.99 : 0;
   276→   276→   775→   775→   775→  const oversLeft = ballsLeft / 6;

   351→   351→   850→   850→   850→export function generateRoundRobinSchedule(teamIds: string[]): [string, string][] {
   352→   352→   851→   851→   851→  const n = teamIds.length;
   353→   353→   852→   852→   852→  const teams = n % 2 === 0 ? [...teamIds] : [...teamIds, 'BYE'];
   354→   354→   853→   853→   853→  const rounds: [string, string][] = [];
   355→   355→   854→   854→   854→  const t = teams.length;
   356→   356→   855→   855→   855→  const pivot = teams[0];
   357→   357→   856→   856→   856→  const rotatable = teams.slice(1);
   358→   358→   857→   857→   857→  for (let round = 0; round < t - 1; round++) {
   359→   359→   858→   858→   858→    const current = [pivot, ...rotatable];
   360→   360→   859→   859→   859→    for (let i = 0; i < t / 2; i++) {
   361→   361→   860→   860→   860→      if (current[i] !== 'BYE' && current[t - 1 - i] !== 'BYE') {
   362→   362→   861→   861→   861→        rounds.push([current[i], current[t - 1 - i]]);
   363→   363→   862→   862→   862→      }
   364→   364→   863→   863→   863→    }
   365→   365→   864→   864→   864→    rotatable.unshift(rotatable.pop()!);
   366→   366→   865→   865→   865→  }
   367→   367→   866→   866→   866→  return rounds;
   368→   368→   867→   867→   867→}
