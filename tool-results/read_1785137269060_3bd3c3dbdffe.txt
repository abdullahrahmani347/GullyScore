    49→    49→function ScrollProgress() {
    50→    50→  const barRef = useRef<HTMLDivElement>(null);
    51→    51→  useLayoutEffect(() => {
    52→    52→    const st = ScrollTrigger.create({
    53→    53→      start: 0,
    54→    54→      end: 'max',
    55→    55→      onUpdate: (self) => {
    56→    56→        if (barRef.current) {
    57→    57→          barRef.current.style.transform = `scaleX(${self.progress})`;
    58→    58→        }
    59→    59→      },
    60→    60→    });
    61→    61→    return () => st.kill();
    62→    62→  }, []);
    63→    63→  return (
    64→    64→    <div className="fixed top-0 left-0 right-0 h-[2px] z-50 bg-bg-card/40">
    65→    65→      <div
    66→    66→        ref={barRef}
    67→    67→        className="h-full bg-accent origin-left"
    68→    68→        style={{ transform: 'scaleX(0)', boxShadow: '0 0 12px rgba(0,212,170,0.6)' }}
