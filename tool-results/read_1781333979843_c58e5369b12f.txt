   452→   452→export async function recordBall(
   453→   453→  prisma: PrismaClient,
   454→   454→  inningsId: string,
   455→   455→  input: RecordBallInput
   456→   456→): Promise<RecordBallResponse> {
   457→   457→  const innings = await prisma.innings.findUniqueOrThrow({
   458→   458→    where: { id: inningsId },
   459→   459→    include: {
   460→   460→      match: true,
   461→   461→      batting: true,
   462→   462→      bowling: true,
   463→   463→      balls: { orderBy: { deliveryNumber: 'desc' }, take: 1 },
   464→   464→    },
   465→   465→  });
   466→   466→
   467→   467→  const {
   468→   468→    batsmanId, bowlerId, runs, isWicket, wicketType,
   469→   469→    dismissedPlayerId, fielderPlayerId, extraType, extraRuns,
   470→   470→  } = input;
   471→   471→
   472→   472→  const isWide = extraType === 'WIDE';
   473→   473→  const isNoBall = extraType === 'NO_BALL';
   474→   474→  const isBye = extraType === 'BYE';
   475→   475→  const isLegBye = extraType === 'LEG_BYE';
   476→   476→  const isLegalDelivery = !isWide && !isNoBall;
   477→   477→  const totalRuns = runs + extraRuns;
   478→   478→
   479→   479→  // runs credited to bowler (byes/legbyes are NOT bowler's fault)
   480→   480→  const runsAgainstBowler = (isBye || isLegBye) ? 0 : totalRuns;
   481→   481→
   482→   482→  const currentLegalBalls = innings.currentBalls;
   483→   483→  const newBallInOver = isLegalDelivery ? currentLegalBalls + 1 : 0;
   484→   484→  const isOverComplete = isLegalDelivery && newBallInOver === 6;
   485→   485→
   486→   486→  const lastDelivery = innings.balls[0];
   487→   487→  const nextDeliveryNumber = (lastDelivery?.deliveryNumber ?? 0) + 1;
   488→   488→
   489→   489→  // ── 1. Create Ball record ─────────────────────────────────
   490→   490→  const ball = await prisma.ball.create({
   491→   491→    data: {
   492→   492→      inningsId,
   493→   493→      overNumber: innings.completedOvers,
   494→   494→      ballInOver: isLegalDelivery ? newBallInOver : 0,
   495→   495→      deliveryNumber: nextDeliveryNumber,
   496→   496→      batsmanId,
   497→   497→      bowlerId,
   498→   498→      runs,
   499→   499→      isWicket,
   500→   500→      wicketType: isWicket ? wicketType : null,
   501→   501→      dismissedPlayerId: isWicket ? dismissedPlayerId : null,
   502→   502→      fielderPlayerId: (isWicket && fielderPlayerId) ? fielderPlayerId : null,
   503→   503→      extraType: extraType ?? null,
   504→   504→      extraRuns,
   505→   505→      isLegalDelivery,
   506→   506→      strikerIdBefore: innings.strikerId!,
   507→   507→      nonStrikerIdBefore: innings.nonStrikerId!,
   508→   508→    },
   509→   509→  });
   510→   510→
   511→   511→  // ── 2. Update BatsmanInnings (striker) ───────────────────
   512→   512→  const batsmanDismissed = isWicket && (
   513→   513→    wicketType !== 'RUN_OUT'
   514→   514→      ? true
   515→   515→      : dismissedPlayerId === batsmanId
   516→   516→  );
   517→   517→
   518→   518→  await prisma.batsmanInnings.upsert({
   519→   519→    where: { inningsId_playerId: { inningsId, playerId: batsmanId } },
   520→   520→    create: {
   521→   521→      inningsId, playerId: batsmanId, battingOrder: 99,
   522→   522→      runs: runs,
   523→   523→      balls: isWide ? 0 : 1,
   524→   524→      fours: runs === 4 && !isBye && !isLegBye ? 1 : 0,
   525→   525→      sixes: runs === 6 && !isBye && !isLegBye ? 1 : 0,
   526→   526→      isOut: batsmanDismissed,
   527→   527→      dismissalType: batsmanDismissed ? wicketType : null,
   528→   528→      dismissedByBowlerId: batsmanDismissed && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? bowlerId : null,
   529→   529→      fielderPlayerId: batsmanDismissed && fielderPlayerId ? fielderPlayerId : null,
   530→   530→    },
   531→   531→    update: {
   532→   532→      runs: { increment: runs },
   533→   533→      balls: { increment: isWide ? 0 : 1 },
   534→   534→      fours: { increment: runs === 4 && !isBye && !isLegBye ? 1 : 0 },
   535→   535→      sixes: { increment: runs === 6 && !isBye && !isLegBye ? 1 : 0 },
   536→   536→      ...(batsmanDismissed ? {
   537→   537→        isOut: true,
   538→   538→        dismissalType: wicketType,
   539→   539→        dismissedByBowlerId: !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? bowlerId : null,
   540→   540→        fielderPlayerId: fielderPlayerId ?? null,
   541→   541→      } : {}),
   542→   542→    },
   543→   543→  });
   544→   544→
   545→   545→  // Handle run-out of non-striker
   546→   546→  if (isWicket && wicketType === 'RUN_OUT' && dismissedPlayerId === innings.nonStrikerId) {
   547→   547→    await prisma.batsmanInnings.upsert({
   548→   548→      where: { inningsId_playerId: { inningsId, playerId: innings.nonStrikerId! } },
   549→   549→      create: {
   550→   550→        inningsId, playerId: innings.nonStrikerId!, battingOrder: 99,
   551→   551→        isOut: true, dismissalType: 'RUN_OUT',
   552→   552→      },
   553→   553→      update: { isOut: true, dismissalType: 'RUN_OUT' },
   554→   554→    });
   555→   555→  }
   556→   556→
   557→   557→  // ── 3. Update BowlerInnings ───────────────────────────────
   558→   558→  const currentBowler = innings.bowling.find(b => b.playerId === bowlerId);
   559→   559→  const bowlerCurrentBalls = currentBowler?.balls ?? 0;
   560→   560→  const bowlerNewBalls = isLegalDelivery ? (bowlerCurrentBalls + 1) % 6 : bowlerCurrentBalls;
   561→   561→  const bowlerOverComplete = isLegalDelivery && (bowlerCurrentBalls + 1) === 6;
   562→   562→
   563→   563→  await prisma.bowlerInnings.upsert({
   564→   564→    where: { inningsId_playerId: { inningsId, playerId: bowlerId } },
   565→   565→    create: {
   566→   566→      inningsId, playerId: bowlerId,
   567→   567→      completedOvers: bowlerOverComplete ? 1 : 0,
   568→   568→      balls: bowlerNewBalls,
   569→   569→      runs: runsAgainstBowler,
   570→   570→      wickets: isWicket && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? 1 : 0,
   571→   571→      wides: isWide ? 1 : 0,
   572→   572→      noBalls: isNoBall ? 1 : 0,
   573→   573→    },
   574→   574→    update: {
   575→   575→      balls: bowlerNewBalls,
   576→   576→      runs: { increment: runsAgainstBowler },
   577→   577→      ...(bowlerOverComplete ? { completedOvers: { increment: 1 } } : {}),
   578→   578→      ...(isWicket && !['RUN_OUT','RETIRED_HURT'].includes(wicketType!) ? { wickets: { increment: 1 } } : {}),
   579→   579→      ...(isWide ? { wides: { increment: 1 } } : {}),
   580→   580→      ...(isNoBall ? { noBalls: { increment: 1 } } : {}),
   581→   581→    },
   582→   582→  });
   583→   583→
   584→   584→  // ── 4. Update Innings state ───────────────────────────────
   585→   585→  const newCurrentBalls = isOverComplete ? 0 : (isLegalDelivery ? currentLegalBalls + 1 : currentLegalBalls);
   586→   586→  const newCompletedOvers = innings.completedOvers + (isOverComplete ? 1 : 0);
   587→   587→  const newWickets = innings.wickets + (isWicket ? 1 : 0);
   588→   588→  const newRuns = innings.runs + totalRuns;
   589→   589→
   590→   590→  // ── 5. Calculate new striker ─────────────────────────────
   591→   591→  let newStrikerId = innings.strikerId!;
   592→   592→  let newNonStrikerId = innings.nonStrikerId!;
   593→   593→
   594→   594→  if (isLegalDelivery) {
   595→   595→    if (runs % 2 === 1) {
   596→   596→      [newStrikerId, newNonStrikerId] = [newNonStrikerId, newStrikerId];
   597→   597→    }
   598→   598→    if (isOverComplete) {
   599→   599→      // At end of over: if even runs, must also rotate (non-striker faces next over)
   600→   600→      if (runs % 2 === 0) {
   601→   601→        [newStrikerId, newNonStrikerId] = [newNonStrikerId, newStrikerId];
   602→   602→      }
   603→   603→    }
   604→   604→  }
   605→   605→
   606→   606→  // If striker dismissed, non-striker takes strike position while new batsman awaits
   607→   607→  if (batsmanDismissed) {
   608→   608→    newStrikerId = newNonStrikerId;
   609→   609→    newNonStrikerId = ''; // will be filled by new batsman selection
   610→   610→  }
   611→   611→
   612→   612→  // ── 6. Persist innings state ──────────────────────────────
   613→   613→  await prisma.innings.update({
   614→   614→    where: { id: inningsId },
   615→   615→    data: {
   616→   616→      runs: newRuns,
   617→   617→      wickets: newWickets,
   618→   618→      completedOvers: newCompletedOvers,
   619→   619→      currentBalls: newCurrentBalls,
   620→   620→      wideBalls: { increment: isWide ? 1 : 0 },
   621→   621→      noBalls: { increment: isNoBall ? 1 : 0 },
   622→   622→      byes: { increment: isBye ? extraRuns : 0 },
   623→   623→      legByes: { increment: isLegBye ? extraRuns : 0 },
   624→   624→      strikerId: newStrikerId || null,
   625→   625→      nonStrikerId: newNonStrikerId || null,
   626→   626→    },
   627→   627→  });
   628→   628→
   629→   629→  // ── 7. Check completion ────────────────────────────────────
   630→   630→  const match = innings.match;
   631→   631→  const isInningsComplete =
   632→   632→    newWickets >= match.maxWickets ||
   633→   633→    (newCompletedOvers >= match.totalOvers && newCurrentBalls === 0) ||
   634→   634→    (innings.inningsNumber === 2 && innings.target != null && newRuns >= innings.target);
   635→   635→
   636→   636→  const ballsRemaining = innings.inningsNumber === 2 && innings.target != null
   637→   637→    ? (match.totalOvers * 6) - (newCompletedOvers * 6 + newCurrentBalls)
   638→   638→    : null;
   639→   639→
   640→   640→  return {
   641→   641→    ball: ball as any,
   642→   642→    inningsState: {
   643→   643→      runs: newRuns,
   644→   644→      wickets: newWickets,
   645→   645→      completedOvers: newCompletedOvers,
   646→   646→      currentBalls: newCurrentBalls,
   647→   647→      currentRunRate: calculateCRR(newRuns, newCompletedOvers, newCurrentBalls),
   648→   648→      requiredRunRate: innings.target
   649→   649→        ? calculateRRR(innings.target - newRuns, match.totalOvers, newCompletedOvers, newCurrentBalls)
   650→   650→        : null,
   651→   651→      runsNeeded: innings.target ? innings.target - newRuns : null,
   652→   652→      ballsRemaining,
   653→   653→      isCompleted: isInningsComplete,
   654→   654→      isOverComplete,
   655→   655→    },
   656→   656→    strikerUpdate: { strikerId: newStrikerId, nonStrikerId: newNonStrikerId },
   657→   657→    needsNewBatsman: isWicket && !isInningsComplete,
   658→   658→    needsNewBowler: isOverComplete && !isInningsComplete,
   659→   659→    needsInningsBreak: isInningsComplete && innings.inningsNumber === 1,
   660→   660→    isMatchComplete: isInningsComplete && innings.inningsNumber === 2,
   661→   661→  };
   662→   662→}

   664→   664→export async function undoLastBall(
   665→   665→  prisma: PrismaClient,
   666→   666→  inningsId: string
   667→   667→): Promise<{ success: boolean }> {
   668→   668→  // 1. Get last ball
   669→   669→  const lastBall = await prisma.ball.findFirst({
   670→   670→    where: { inningsId },
   671→   671→    orderBy: { deliveryNumber: 'desc' },
   672→   672→  });
   673→   673→  if (!lastBall) return { success: false };
   674→   674→
   675→   675→  const innings = await prisma.innings.findUniqueOrThrow({ where: { id: inningsId }, include: { match: true } });
   676→   676→
   677→   677→  const isWide = lastBall.extraType === 'WIDE';
   678→   678→  const isNoBall = lastBall.extraType === 'NO_BALL';
   679→   679→  const isBye = lastBall.extraType === 'BYE';
   680→   680→  const isLegBye = lastBall.extraType === 'LEG_BYE';
   681→   681→  const totalRuns = lastBall.runs + lastBall.extraRuns;
   682→   682→  const runsAgainstBowler = (isBye || isLegBye) ? 0 : totalRuns;
   683→   683→
   684→   684→  // 2. Reverse BatsmanInnings
   685→   685→  await prisma.batsmanInnings.update({
   686→   686→    where: { inningsId_playerId: { inningsId, playerId: lastBall.batsmanId } },
   687→   687→    data: {
   688→   688→      runs: { decrement: lastBall.runs },
   689→   689→      balls: { decrement: isWide ? 0 : 1 },
   690→   690→      fours: { decrement: lastBall.runs === 4 && !isBye && !isLegBye ? 1 : 0 },
   691→   691→      sixes: { decrement: lastBall.runs === 6 && !isBye && !isLegBye ? 1 : 0 },
   692→   692→      ...(lastBall.isWicket ? { isOut: false, dismissalType: null, dismissedByBowlerId: null, fielderPlayerId: null } : {}),
   693→   693→    },
   694→   694→  });
   695→   695→
   696→   696→  // Reverse non-striker run-out
   697→   697→  if (lastBall.isWicket && lastBall.wicketType === 'RUN_OUT' && lastBall.dismissedPlayerId === lastBall.nonStrikerIdBefore) {
   698→   698→    await prisma.batsmanInnings.update({
   699→   699→      where: { inningsId_playerId: { inningsId, playerId: lastBall.nonStrikerIdBefore } },
   700→   700→      data: { isOut: false, dismissalType: null },
   701→   701→    });
   702→   702→  }
   703→   703→
   704→   704→  // 3. Reverse BowlerInnings
   705→   705→  const bowlerInnings = await prisma.bowlerInnings.findUnique({
   706→   706→    where: { inningsId_playerId: { inningsId, playerId: lastBall.bowlerId } },
   707→   707→  });
   708→   708→  if (bowlerInnings) {
   709→   709→    const wasOverBall = lastBall.isLegalDelivery && lastBall.ballInOver === 6;
   710→   710→    await prisma.bowlerInnings.update({
   711→   711→      where: { inningsId_playerId: { inningsId, playerId: lastBall.bowlerId } },
   712→   712→      data: {
   713→   713→        runs: { decrement: runsAgainstBowler },
   714→   714→        balls: wasOverBall ? 0 : { decrement: lastBall.isLegalDelivery ? 1 : 0 },
   715→   715→        ...(wasOverBall ? { completedOvers: { decrement: 1 }, balls: 5 } : {}),
   716→   716→        ...(lastBall.isWicket && !['RUN_OUT','RETIRED_HURT'].includes(lastBall.wicketType ?? '') ? { wickets: { decrement: 1 } } : {}),
   717→   717→        ...(isWide ? { wides: { decrement: 1 } } : {}),
   718→   718→        ...(isNoBall ? { noBalls: { decrement: 1 } } : {}),
   719→   719→      },
   720→   720→    });
   721→   721→  }
   722→   722→
   723→   723→  // 4. Reverse Innings
   724→   724→  const wasOverComplete = lastBall.isLegalDelivery && lastBall.ballInOver === 6;
   725→   725→  const restoredBalls = wasOverComplete ? 5 : (lastBall.isLegalDelivery ? Math.max(0, innings.currentBalls - 1) : innings.currentBalls);
   726→   726→  const restoredCompletedOvers = innings.completedOvers - (wasOverComplete ? 1 : 0);
   727→   727→
   728→   728→  await prisma.innings.update({
   729→   729→    where: { id: inningsId },
   730→   730→    data: {
   731→   731→      runs: { decrement: totalRuns },
   732→   732→      wickets: { decrement: lastBall.isWicket ? 1 : 0 },
   733→   733→      completedOvers: restoredCompletedOvers,
   734→   734→      currentBalls: restoredBalls,
   735→   735→      wideBalls: { decrement: isWide ? 1 : 0 },
   736→   736→      noBalls: { decrement: isNoBall ? 1 : 0 },
   737→   737→      byes: { decrement: isBye ? lastBall.extraRuns : 0 },
   738→   738→      legByes: { decrement: isLegBye ? lastBall.extraRuns : 0 },
   739→   739→      strikerId: lastBall.strikerIdBefore,
   740→   740→      nonStrikerId: lastBall.nonStrikerIdBefore,
   741→   741→    },
   742→   742→  });
   743→   743→
   744→   744→  // 5. Delete Ball record
   745→   745→  await prisma.ball.delete({ where: { id: lastBall.id } });
   746→   746→
   747→   747→  return { success: true };
   748→   748→}

   755→   755→export function decimalOvers(completedOvers: number, balls: number): number {
   756→   756→  return completedOvers + balls / 6;

   759→   759→export function calculateCRR(runs: number, completedOvers: number, balls: number): number {
   760→   760→  const ov = decimalOvers(completedOvers, balls);
   761→   761→  if (ov === 0) return 0;
   762→   762→  return Math.round((runs / ov) * 100) / 100;
   763→   763→}
   764→   764→
   765→   765→export function calculateRRR(
   766→   766→  runsNeeded: number,
   767→   767→  totalOvers: number,
   768→   768→  completedOvers: number,
   769→   769→  balls: number
   770→   770→): number {
   771→   771→  const ballsTotal = totalOvers * 6;
   772→   772→  const ballsUsed = completedOvers * 6 + balls;
   773→   773→  const ballsLeft = ballsTotal - ballsUsed;
   774→   774→  if (ballsLeft <= 0) return runsNeeded > 0 ? 99.99 : 0;
   775→   775→  const oversLeft = ballsLeft / 6;
   776→   776→  return Math.round((runsNeeded / oversLeft) * 100) / 100;
   777→   777→}

   803→   803→export function calculateNRR(
   804→   804→  runsScored: number,
   805→   805→  oversFaced: number,

   807→   807→  oversBowled: number

   848→   848→}

   850→   850→export function generateRoundRobinSchedule(teamIds: string[]): [string, string][] {
   851→   851→  const n = teamIds.length;
   852→   852→  const teams = n % 2 === 0 ? [...teamIds] : [...teamIds, 'BYE'];
   853→   853→  const rounds: [string, string][] = [];
   854→   854→  const t = teams.length;
   855→   855→  const pivot = teams[0];
   856→   856→  const rotatable = teams.slice(1);
   857→   857→  for (let round = 0; round < t - 1; round++) {
   858→   858→    const current = [pivot, ...rotatable];
   859→   859→    for (let i = 0; i < t / 2; i++) {
   860→   860→      if (current[i] !== 'BYE' && current[t - 1 - i] !== 'BYE') {
   861→   861→        rounds.push([current[i], current[t - 1 - i]]);
   862→   862→      }
   863→   863→    }
   864→   864→    rotatable.unshift(rotatable.pop()!);
   865→   865→  }
   866→   866→  return rounds;
   867→   867→}
