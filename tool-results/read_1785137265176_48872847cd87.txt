    14→ *   <StatsSection />       — pinned, scroll 0.71–0.86 (3D: particles only)
    15→ *   <CTASection />         — pinned, scroll 0.86–1.0 (3D: particles + logo converge)
    16→ *
    17→ * GSAP setup:
    18→ *   - ScrollTrigger.create() with start:0 / end:'max' updates scrollRef.current
    19→ *     on every scroll. The 3D scene reads this ref via useFrame — no React
    20→ *     re-renders, no perf hit.
    21→ *   - Each .pin-section is pinned for +=120% of viewport with scrub:1 for
    22→ *     smooth, slow cinematic pacing.
    23→ *   - Section-internal animations (score updates, chart draws, counter ticks)
    24→ *     are wired to the pin's timeline via gsap.timeline().

    26→ * All content lives above the canvas (relative z-10) with semi-transparent
    27→ * dark backgrounds so the 3D remains visible but text stays readable.

    46→// ─────────────────────────────────────────────────────────────────────
    47→// Scroll progress bar (top of page)
    48→// ─────────────────────────────────────────────────────────────────────
    49→function ScrollProgress() {
    50→  const barRef = useRef<HTMLDivElement>(null);
    51→  useLayoutEffect(() => {
    52→    const st = ScrollTrigger.create({
    53→      start: 0,
    54→      end: 'max',
    55→      onUpdate: (self) => {
    56→        if (barRef.current) {
    57→          barRef.current.style.transform = `scaleX(${self.progress})`;
    58→        }
    59→      },
    60→    });
    61→    return () => st.kill();
    62→  }, []);
    63→  return (
    64→    <div className="fixed top-0 left-0 right-0 h-[2px] z-50 bg-bg-card/40">
    65→      <div
    66→        ref={barRef}
    67→        className="h-full bg-accent origin-left"
    68→        style={{ transform: 'scaleX(0)', boxShadow: '0 0 12px rgba(0,212,170,0.6)' }}
    69→      />
    70→    </div>
    71→  );
    72→}

   850→export default function LandingPage() {
   851→  const containerRef = useRef<HTMLDivElement>(null);
   852→  const scrollRef = useRef(0);
   853→  const [mounted, setMounted] = useState(false);
   854→
   855→  useEffect(() => {
   856→    setMounted(true);
   857→  }, []);
   858→
   859→  useLayoutEffect(() => {
   860→    if (!mounted) return;
   861→
   862→    const ctx = gsap.context(() => {
   863→      // ── Global scroll progress for 3D scene ──
   864→      ScrollTrigger.create({
   865→        start: 0,
   866→        end: 'max',
   867→        onUpdate: (self) => {
   868→          scrollRef.current = self.progress;
   869→        },
   870→      });
   871→
   872→      // ── Hero section: entrance plays on mount (NOT scrubbed), exit is scrubbed ──
   873→      // Using .from() with scrub would leave content at opacity:0 at scroll 0,
   874→      // making the hero blank on page load. Instead, play the entrance on mount
   875→      // and use a separate scrubbed timeline for the scroll exit.
   876→      gsap.timeline({ delay: 0.4 })
   877→        .from('.hero-badge', { y: 20, opacity: 0, duration: 1, ease: 'power3.out' })
   878→        .from('.hero-title', { y: 30, opacity: 0, duration: 1.2, ease: 'power3.out' }, '-=0.6')
   879→        .from('.hero-tagline', { y: 20, opacity: 0, duration: 1, ease: 'power3.out' }, '-=0.7')
   880→        .from('.hero-cta', { y: 20, opacity: 0, duration: 1, ease: 'power3.out' }, '-=0.6')
   881→        .from('.hero-scroll-indicator', { opacity: 0, duration: 1, ease: 'power2.out' }, '-=0.4');
   882→
   883→      // Hero scroll exit (scrubbed + pinned)
   884→      gsap.timeline({
   885→        scrollTrigger: {
   886→          trigger: '#hero',
   887→          start: 'top top',
   888→          end: '+=120%',
   889→          pin: true,
   890→          scrub: 1,
   891→        },
   892→      }).to('.hero-content', { opacity: 0, y: -80, duration: 1, ease: 'none' });
   893→
   894→      // ── Live scoring: phone score ticks up, balls animate in ──
   895→      const liveTl = gsap.timeline({
   896→        scrollTrigger: {
   897→          trigger: '#live-scoring',
   898→          start: 'top top',
   899→          end: '+=120%',
   900→          pin: true,
   901→          scrub: 1,
   902→        },
   903→      });
   904→      // Animate feature cards in
   905→      liveTl.from('.live-feature-card', {
   906→        y: 40,
   907→        opacity: 0,
