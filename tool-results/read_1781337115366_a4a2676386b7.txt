     5→// --- Seeded random for reproducibility ---
     6→function seededRandom(seed: number) {
     7→  let s = seed;
     8→  return () => {
     9→    s = (s * 16807 + 0) % 2147483647;
    10→    return (s - 1) / 2147483646;
    11→  };
    12→}

    14→// --- Types ---
    15→interface SeedBall {
    16→  runs: number;
    17→  extraType?: 'WIDE' | 'NO_BALL' | 'BYE' | 'LEG_BYE';
    18→  extraRuns?: number;
    19→  isWicket?: boolean;
    20→  wicketType?: 'BOWLED' | 'CAUGHT' | 'RUN_OUT' | 'LBW' | 'STUMPED' | 'HIT_WICKET' | 'RETIRED_HURT';
    21→  batsmanIndex: number; // Index into batsmen array
    22→  bowlerIndex: number;  // Index into bowlers array
    23→  dismissedBatsmanIndex?: number; // Which batsman got out (for run outs)
    24→}
    25→
    26→interface InningsSpec {
    27→  teamBattingId: string;
    28→  teamBowlingId: string;
    29→  totalRuns: number;
    30→  totalWickets: number;
    31→  totalOvers: number;
    32→  batsmen: string[]; // Player IDs in batting order
    33→  bowlers: string[]; // Player IDs who bowl
    34→  balls: SeedBall[];
    35→}

    37→// Helper: generate a realistic ball sequence for an innings
    38→function generateInningsBalls(
    39→  totalRuns: number,
    40→  totalWickets: number,
    41→  totalOvers: number,
    42→  numBatsmen: number,
    43→  numBowlers: number,
    44→  seed: number
    45→): SeedBall[] {
    46→  const rand = seededRandom(seed);
    47→  const balls: SeedBall[] = [];
    48→  let runsSoFar = 0;
    49→  let wicketsSoFar = 0;
    50→  let legalDeliveries = 0;
    51→  const maxLegalDeliveries = totalOvers * 6;
    52→  let currentBatsman = 0; // Index in batsmen array
    53→  let currentBowler = 0;
    54→  let ballInOver = 0;
    55→  let batsmenOut = new Set<number>();
    56→
    57→  while (legalDeliveries < maxLegalDeliveries && wicketsSoFar < totalWickets) {
    58→    const runsLeft = totalRuns - runsSoFar;
    59→    const ballsLeft = maxLegalDeliveries - legalDeliveries;
    60→    const wicketsLeft = totalWickets - wicketsSoFar;
    61→
    62→    // Determine this ball outcome based on probabilities and targets
    63→    const r = rand();
    64→
    65→    let ball: SeedBall;
    66→
    67→    if (r < 0.03 && wicketsLeft > 0) {
    68→      // Wicket (3% per ball)
    69→      const wicketTypes = ['BOWLED', 'CAUGHT', 'LBW', 'RUN_OUT', 'STUMPED'] as const;
    70→      const wt = wicketTypes[Math.floor(rand() * wicketTypes.length)];
    71→      
    72→      ball = {
    73→        runs: wt === 'RUN_OUT' ? Math.floor(rand() * 2) : 0,
    74→        isWicket: true,
    75→        wicketType: wt,
    76→        batsmanIndex: currentBatsman,
    77→        bowlerIndex: currentBowler,
    78→        dismissedBatsmanIndex: wt === 'RUN_OUT' ? currentBatsman : currentBatsman,
    79→      };
    80→      wicketsSoFar++;
    81→      batsmenOut.add(currentBatsman);
    82→      // Move to next batsman
    83→      currentBatsman = Math.min(currentBatsman + 1, numBatsmen - 1);
    84→      legalDeliveries++;
    85→    } else if (r < 0.06) {
    86→      // Wide (3%)
    87→      const wideRuns = rand() < 0.85 ? 1 : (rand() < 0.5 ? 2 : 3);
    88→      ball = {
    89→        runs: 0,
    90→        extraType: 'WIDE',
    91→        extraRuns: wideRuns,
    92→        batsmanIndex: currentBatsman,
    93→        bowlerIndex: currentBowler,
    94→      };
    95→      runsSoFar += wideRuns;
    96→      // Wide doesn't count as legal delivery, don't increment legalDeliveries
    97→    } else if (r < 0.08) {
    98→      // No ball (2%)
    99→      const nbRuns = rand() < 0.7 ? 1 : 0;
   100→      const batsmanRuns = nbRuns > 0 ? 0 : Math.floor(rand() * 4);
   101→      ball = {
   102→        runs: batsmanRuns,
   103→        extraType: 'NO_BALL',
   104→        extraRuns: 1 + (batsmanRuns > 0 ? 0 : 0),
   105→        batsmanIndex: currentBatsman,
   106→        bowlerIndex: currentBowler,
   107→      };
   108→      runsSoFar += 1 + batsmanRuns;
   109→      // No ball doesn't count as legal delivery
   110→    } else if (r < 0.10 && runsLeft > 0) {
   111→      // Bye/Leg bye (2%)
   112→      const byeRuns = Math.floor(rand() * 3) + 1;
   113→      ball = {
   114→        runs: 0,
   115→        extraType: rand() < 0.5 ? 'BYE' : 'LEG_BYE',
   116→        extraRuns: byeRuns,
   117→        batsmanIndex: currentBatsman,
   118→        bowlerIndex: currentBowler,
   119→      };
   120→      runsSoFar += byeRuns;
   121→      legalDeliveries++;
   122→    } else {
   123→      // Normal ball - distribute runs realistically
   124→      let runs: number;
   125→      const rr = rand();
   126→      if (rr < 0.38) runs = 0;       // 38% dots
   127→      else if (rr < 0.63) runs = 1;  // 25% singles
   128→      else if (rr < 0.71) runs = 2;  // 8% twos
   129→      else if (rr < 0.74) runs = 3;  // 3% threes
   130→      else if (rr < 0.87) runs = 4;  // 13% fours
   131→      else runs = 6;                  // 6% sixes
   132→
   133→      ball = {
   134→        runs,
   135→        batsmanIndex: currentBatsman,
   136→        bowlerIndex: currentBowler,
   137→      };
   138→      runsSoFar += runs;
   139→      legalDeliveries++;
   140→
   141→      // Rotate strike on odd runs
   142→      if (runs % 2 === 1) {
   143→        // Striker changes at end of over or on odd run
   144→        // We handle rotation in the seedInnings function
   145→      }
   146→    }
   147→
   148→    balls.push(ball);
   149→
   150→    // Handle over change - rotate bowler
   151→    ballInOver++;
   152→    if (ballInOver >= 6 && legalDeliveries % 6 === 0) {
   153→      ballInOver = 0;
   154→      // Switch bowler (alternate between 2-3 bowlers)
   155→      currentBowler = (currentBowler + 1) % numBowlers;
   156→    }
   157→  }
   158→
   159→  return balls;
   160→}

   162→// Seed an innings with ball-by-ball data
   163→async function seedInnings(
   164→  prisma: PrismaClient,
   165→  matchId: string,
   166→  inningsNumber: number,
   167→  spec: InningsSpec,
   168→  target?: number
   169→) {
   170→  const batsmen = spec.batsmen;
   171→  const bowlers = spec.bowlers;
   172→  const balls = spec.balls;
   173→
   174→  // Create innings record
   175→  const innings = await prisma.innings.create({
   176→    data: {
   177→      matchId,
   178→      teamId: spec.teamBattingId,
   179→      inningsNumber,
   180→      target: target || null,
   181→    },
   182→  });
   183→
   184→  // Track state
   185→  let totalRuns = 0;
   186→  let totalWickets = 0;
   187→  let legalDeliveries = 0;
   188→  let overNumber = 0;
   189→  let ballInOver = 0;
   190→  let deliveryNumber = 0;
   191→  let strikerIdx = 0; // Index into batsmen array (0 = opener 1)
   192→  let nonStrikerIdx = 1; // Index into batsmen array (1 = opener 2)
   193→  let nextBatsmanIdx = 2; // Next batsman to come in
   194→
   195→  // Track batsman innings
   196→  const batsmanInningsMap = new Map<string, string>(); // player ID -> batsmanInnings ID
   197→  const bowlerInningsMap = new Map<string, string>(); // player ID -> bowlerInnings ID
   198→
   199→  // Create opening batsman innings
   200→  for (const pId of [batsmen[0], batsmen[1]]) {
   201→    const bi = await prisma.batsmanInnings.create({
   202→      data: {
   203→        inningsId: innings.id,
   204→        playerId: pId,
   205→        battingOrder: batsmen.indexOf(pId),
   206→      },
   207→    });
   208→    batsmanInningsMap.set(pId, bi.id);
   209→  }
   210→
   211→  for (const ball of balls) {
   212→    const batsmanId = batsmen[ball.batsmanIndex] || batsmen[strikerIdx];
   213→    const bowlerId = bowlers[ball.bowlerIndex] || bowlers[0];
   214→    const isLegal = ball.extraType !== 'WIDE' && ball.extraType !== 'NO_BALL';
   215→
   216→    // Ensure batsman innings exists
   217→    if (!batsmanInningsMap.has(batsmanId)) {
   218→      const bi = await prisma.batsmanInnings.create({
   219→        data: {
   220→          inningsId: innings.id,
   221→          playerId: batsmanId,
   222→          battingOrder: batsmen.indexOf(batsmanId),
   223→        },
   224→      });
   225→      batsmanInningsMap.set(batsmanId, bi.id);
   226→    }
   227→
   228→    // Ensure bowler innings exists
   229→    if (!bowlerInningsMap.has(bowlerId)) {
   230→      const boi = await prisma.bowlerInnings.create({
   231→        data: {
   232→          inningsId: innings.id,
   233→          playerId: bowlerId,
   234→        },
   235→      });
   236→      bowlerInningsMap.set(bowlerId, boi.id);
   237→    }
   238→
   239→    deliveryNumber++;
   240→
   241→    // Create ball record
   242→    await prisma.ball.create({
   243→      data: {
   244→        inningsId: innings.id,
   245→        overNumber,
   246→        ballInOver,
   247→        deliveryNumber,
   248→        batsmanId,
   249→        bowlerId,
   250→        runs: ball.runs,
   251→        isWicket: ball.isWicket || false,
   252→        wicketType: ball.wicketType || null,
   253→        dismissedPlayerId: ball.isWicket ? batsmanId : null,
   254→        fielderPlayerId:
   255→          ball.isWicket && ball.wicketType === 'CAUGHT'
   256→            ? bowlers[(ball.bowlerIndex + 1) % bowlers.length]
   257→            : ball.isWicket && ball.wicketType === 'RUN_OUT'
   258→            ? batsmen[ball.dismissedBatsmanIndex !== undefined ? ball.dismissedBatsmanIndex : strikerIdx === nonStrikerIdx ? nonStrikerIdx + 1 : nonStrikerIdx]
   259→            : null,
   260→        extraType: ball.extraType || null,
   261→        extraRuns: ball.extraRuns || 0,
   262→        isLegalDelivery: isLegal,
   263→        strikerIdBefore: batsmen[strikerIdx],
   264→        nonStrikerIdBefore: batsmen[nonStrikerIdx],
   265→      },
   266→    });
   267→
   268→    // Update batsman innings
   269→    const batsmanInningsId = batsmanInningsMap.get(batsmanId)!;
   270→    if (isLegal && ball.extraType !== 'BYE' && ball.extraType !== 'LEG_BYE') {
   271→      await prisma.batsmanInnings.update({
   272→        where: { id: batsmanInningsId },
   273→        data: {
   274→          runs: { increment: ball.runs },
   275→          balls: { increment: 1 },
   276→          fours: { increment: ball.runs === 4 ? 1 : 0 },
   277→          sixes: { increment: ball.runs === 6 ? 1 : 0 },
   278→        },
   279→      });
   280→    } else if (ball.extraType === 'BYE' || ball.extraType === 'LEG_BYE') {
   281→      await prisma.batsmanInnings.update({
   282→        where: { id: batsmanInningsId },
   283→        data: {
   284→          balls: { increment: 1 },
   285→        },
   286→      });
   287→    }
   288→
   289→    // Handle wicket
   290→    if (ball.isWicket) {
   291→      totalWickets++;
   292→      await prisma.batsmanInnings.update({
   293→        where: { id: batsmanInningsId },
   294→        data: {
   295→          isOut: true,
   296→          dismissalType: ball.wicketType || 'BOWLED',
   297→          dismissedByBowlerId:
   298→            ball.wicketType !== 'RUN_OUT' ? bowlerId : null,
   299→          fielderPlayerId:
   300→            ball.wicketType === 'CAUGHT'
   301→              ? bowlers[(ball.bowlerIndex + 1) % bowlers.length]
   302→              : ball.wicketType === 'RUN_OUT'
   303→              ? batsmen[nonStrikerIdx]
   304→              : null,
   305→        },
   306→      });
   307→
   308→      // Bring in next batsman
   309→      if (nextBatsmanIdx < batsmen.length) {
   310→        strikerIdx = nextBatsmanIdx;
   311→        nextBatsmanIdx++;
   312→        // Ensure new batsman innings exists
   313→        const newBatsmanId = batsmen[strikerIdx];
   314→        if (!batsmanInningsMap.has(newBatsmanId)) {
   315→          const bi = await prisma.batsmanInnings.create({
   316→            data: {
   317→              inningsId: innings.id,
   318→              playerId: newBatsmanId,
   319→              battingOrder: strikerIdx,
   320→            },
   321→          });
   322→          batsmanInningsMap.set(newBatsmanId, bi.id);
   323→        }
   324→      }
   325→    }
   326→
   327→    // Update bowler innings
   328→    const bowlerInningsId = bowlerInningsMap.get(bowlerId)!;
   329→    const bowlerUpdate: any = {};
   330→    if (ball.extraType === 'WIDE') {
   331→      bowlerUpdate.wides = { increment: 1 };
   332→      bowlerUpdate.runs = { increment: ball.extraRuns || 1 };
   333→    } else if (ball.extraType === 'NO_BALL') {
   334→      bowlerUpdate.noBalls = { increment: 1 };
   335→      bowlerUpdate.runs = { increment: (ball.extraRuns || 1) + ball.runs };
   336→    } else {
   337→      bowlerUpdate.runs = { increment: ball.runs + (ball.extraRuns || 0) };
   338→    }
   339→    if (ball.isWicket && ball.wicketType !== 'RUN_OUT') {
   340→      bowlerUpdate.wickets = { increment: 1 };
   341→    }
   342→    if (isLegal) {
   343→      bowlerUpdate.balls = { increment: 1 };
   344→    }
   345→    await prisma.bowlerInnings.update({
   346→      where: { id: bowlerInningsId },
   347→      data: bowlerUpdate,
   348→    });
   349→
   350→    // Update innings totals
   351→    totalRuns += ball.runs + (ball.extraRuns || 0);
   352→
   353→    if (isLegal) {
   354→      legalDeliveries++;
   355→      ballInOver++;
   356→
   357→      // Rotate strike on odd runs
   358→      if (ball.runs % 2 === 1 && !ball.isWicket) {
   359→        [strikerIdx, nonStrikerIdx] = [nonStrikerIdx, strikerIdx];
   360→      }
   361→
   362→      // Over complete
   363→      if (ballInOver >= 6) {
   364→        ballInOver = 0;
   365→        overNumber++;
   366→        // Rotate strike at end of over
   367→        [strikerIdx, nonStrikerIdx] = [nonStrikerIdx, strikerIdx];
   368→      }
   369→    }
   370→  }
   371→
   372→  // Update innings record with final totals
   373→  const finalCompletedOvers = Math.floor(legalDeliveries / 6);
   374→  const finalCurrentBalls = legalDeliveries % 6;
   375→
   376→  // Calculate extras
   377→  const widesCount = balls.filter(b => b.extraType === 'WIDE').reduce((sum, b) => sum + (b.extraRuns || 1), 0);
   378→  const noBallsCount = balls.filter(b => b.extraType === 'NO_BALL').reduce((sum, b) => sum + (b.extraRuns || 1), 0);
   379→  const byesCount = balls.filter(b => b.extraType === 'BYE').reduce((sum, b) => sum + (b.extraRuns || 0), 0);
   380→  const legByesCount = balls.filter(b => b.extraType === 'LEG_BYE').reduce((sum, b) => sum + (b.extraRuns || 0), 0);
   381→
   382→  await prisma.innings.update({
   383→    where: { id: innings.id },
   384→    data: {
   385→      runs: totalRuns,
   386→      wickets: totalWickets,
   387→      completedOvers: finalCompletedOvers,
   388→      currentBalls: finalCurrentBalls,
   389→      wideBalls: widesCount,
   390→      noBalls: noBallsCount,
   391→      byes: byesCount,
   392→      legByes: legByesCount,
   393→      strikerId: batsmen[strikerIdx] || null,
   394→      nonStrikerId: batsmen[nonStrikerIdx] || null,
   395→      currentBowlerId: bowlers[0] || null,
   396→      isCompleted: true,
   397→    },
   398→  });
   399→
   400→  return { totalRuns, totalWickets, completedOvers: finalCompletedOvers, currentBalls: finalCurrentBalls, inningsId: innings.id };
   401→}

   403→// Update tournament team stats after a match
   404→async function updateTournamentStats(
   405→  prisma: PrismaClient,
   406→  tournamentId: string,
   407→  team1Id: string,
   408→  team2Id: string,
   409→  team1Runs: number,
   410→  team1Wickets: number,
   411→  team1Overs: number,
   412→  team2Runs: number,
   413→  team2Wickets: number,
   414→  team2Overs: number,
   415→  winnerId: string | null
   416→) {
   417→  // Calculate overs as decimal for NRR
   418→  const team1OversDecimal = team1Overs;
   419→  const team2OversDecimal = team2Overs;
   420→
   421→  for (const { teamId, runsScored, oversFaced, runsConceded, oversBowled, won } of [
   422→    {
   423→      teamId: team1Id,
   424→      runsScored: team1Runs,
   425→      oversFaced: team1OversDecimal,
   426→      runsConceded: team2Runs,
   427→      oversBowled: team2OversDecimal,
   428→      won: winnerId === team1Id,
   429→    },
   430→    {
   431→      teamId: team2Id,
   432→      runsScored: team2Runs,
   433→      oversFaced: team2OversDecimal,
   434→      runsConceded: team1Runs,
   435→      oversBowled: team1OversDecimal,
   436→      won: winnerId === team2Id,
   437→    },
   438→  ]) {
   439→    const existing = await prisma.tournamentTeam.findUnique({
   440→      where: { tournamentId_teamId: { tournamentId, teamId } },
   441→    });
   442→    if (!existing) continue;
   443→
   444→    const nrr = (existing.runsScored + runsScored) / Math.max(existing.oversFaced + oversFaced, 0.1)
   445→      - (existing.runsConceded + runsConceded) / Math.max(existing.oversBowled + oversBowled, 0.1);
   446→
   447→    await prisma.tournamentTeam.update({
   448→      where: { tournamentId_teamId: { tournamentId, teamId } },
   449→      data: {
   450→        played: { increment: 1 },
   451→        won: { increment: won ? 1 : 0 },
   452→        lost: { increment: won ? 0 : 1 },
   453→        points: { increment: won ? 2 : 0 },
   454→        runsScored: { increment: runsScored },
   455→        runsConceded: { increment: runsConceded },
   456→        oversFaced: { increment: oversFaced },
   457→        oversBowled: { increment: oversBowled },
   458→        nrr: Math.round(nrr * 1000) / 1000,
   459→      },
   460→    });
   461→  }
   462→}

   584→  // Adjust runs to match targets exactly
   585→  function adjustBallsToTarget(balls: SeedBall[], targetRuns: number): SeedBall[] {
   586→    let currentRuns = 0;
   587→    for (const b of balls) {
   588→      currentRuns += b.runs + (b.extraRuns || 0);
   589→    }
   590→    // We'll adjust the last few balls to hit the target
   591→    let diff = targetRuns - currentRuns;
   592→    for (let i = balls.length - 1; i >= 0 && diff !== 0; i--) {
   593→      const b = balls[i];
   594→      if (!b.isWicket && !b.extraType) {
   595→        if (diff > 0) {
   596→          const canAdd = Math.min(diff, 6 - b.runs);
   597→          b.runs += canAdd;
   598→          diff -= canAdd;
   599→        } else {
   600→          const canSub = Math.min(-diff, b.runs);
   601→          b.runs -= canSub;
   602→          diff += canSub;
   603→        }
   604→      }
   605→    }
   606→    // If still not exact, adjust extras
   607→    if (diff !== 0) {
   608→      for (let i = balls.length - 1; i >= 0 && diff !== 0; i--) {
   609→        const b = balls[i];
   610→        if (b.extraType === 'WIDE') {
   611→          if (diff > 0) {
   612→            b.extraRuns = (b.extraRuns || 1) + 1;
   613→            diff -= 1;
   614→          }
   615→        }
   616→      }
   617→    }
   618→    return balls;
   619→  }
